// RPG-DEV-005: preserve prefetch work across a canceled document navigation.
// The check lives at the network boundary, after async RSC cache-key hashing.
let isDocumentDeparting = false;
const deferredPrefetchRequests = new Set();

function resumePrefetchRequests() {
    if (!isDocumentDeparting) return;
    isDocumentDeparting = false;
    for (const resume of Array.from(deferredPrefetchRequests)) resume();
}

if (typeof window !== 'undefined') {
    window.addEventListener('beforeunload', () => {
        isDocumentDeparting = true;
    });
    window.addEventListener('pageshow', resumePrefetchRequests);
    // A dismissed beforeunload dialog does not emit pageshow. Resume on the
    // next interaction, without guessing whether the navigation was canceled.
    window.addEventListener('pointerdown', resumePrefetchRequests, { passive: true });
    window.addEventListener('keydown', resumePrefetchRequests);
}

function deferPrefetchRequest(input, init) {
    return new Promise((resolve, reject) => {
        const signal = init.signal;
        const cleanup = () => {
            deferredPrefetchRequests.delete(resume);
            signal?.removeEventListener('abort', abort);
        };
        const resume = () => {
            cleanup();
            // Recheck departure in case an interaction immediately navigated.
            resolve(fetchInternal(input, init));
        };
        const abort = () => {
            cleanup();
            reject(signal.reason);
        };
        deferredPrefetchRequests.add(resume);
        signal?.addEventListener('abort', abort, { once: true });
    });
}
